import React, { useRef, useState, useEffect } from 'react';
import { Canvas, useFrame } from '@react-three/fiber';
import { OrbitControls, Sky, Environment, Stars } from '@react-three/drei';
import * as THREE from 'three';

// 1. الفيلا حقتك 3D
function Villa() {
  return (
    <mesh position={[0, 0, 0]}>
      <boxGeometry args={[4, 2, 4]} />
      <meshStandardMaterial color="#f5f5f5" />
    </mesh>
  );
}

// 2. طبقة الفور دي: مطر + صوت + زمن
function FourDEffects({ isNight, isRaining }) {
  const rainRef = useRef();
  
  useFrame((state, delta) => {
    if (isRaining && rainRef.current) {
      rainRef.current.position.y -= delta * 10; // المطر ينزل
      if (rainRef.current.position.y < -5) rainRef.current.position.y = 5;
    }
  });

  return (
    <>
      <Sky sunPosition={isNight ? [0, -1, 0] : [0, 1, 0]} /> 
      {isNight ? <Stars /> : null}
      
      {isRaining && (
        <mesh ref={rainRef} position={[0, 5, 0]}>
          <boxGeometry args={[0.05, 0.5, 0.05]} />
          <meshBasicMaterial color="lightblue" transparent opacity={0.5} />
        </mesh>
      )}
      
      <audio src="https://cdn.freesound.org/previews/612/612708_10813652-lq.mp3" autoPlay loop /> {/* صوت مطر */}
    </>
  );
}

export default function App() {
  const [isNight, setIsNight] = useState(false);
  const [isRaining, setIsRaining] = useState(false);

  return (
    <div style={{ width: '100vw', height: '100vh' }}>
      <Canvas camera={{ position: [10, 5, 10] }}>
        <ambientLight intensity={isNight ? 0.2 : 0.5} />
        <directionalLight position={[5, 5, 5]} intensity={isNight ? 0.3 : 1} />
        <Environment preset="city" />
        
        <Villa />
        <FourDEffects isNight={isNight} isRaining={isRaining} />
        <OrbitControls />
      </Canvas>

      <div style={{ position: 'absolute', top: 20, left: 20, zIndex: 1 }}>
        <button onClick={() => setIsNight(!isNight)}>
          {isNight ? '☀️ نهار' : '🌙 ليل'}
        </button>
        <button onClick={() => setIsRaining(!isRaining)}>
          {isRaining ? '⛔ وقف المطر' : '🌧️ مطر'}
        </button>
      </div>
    </div>
  );
}
