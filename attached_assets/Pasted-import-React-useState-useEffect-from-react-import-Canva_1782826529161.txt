import React, { useState, useEffect } from 'react';
import { Canvas } from '@react-three/fiber';
import { OrbitControls, Environment, ContactShadows } from '@react-three/drei';
import * as THREE from 'three';

// ================== 1. ربط الـ AI المصمم = API حقي ==================
const TRIPO_API_KEY = "ضع_مفتاحك_هنا"; // جيبه مجاني من tripo3d.ai

async function generate3DFromText(prompt) {
  const res = await fetch("https://api.tripo3d.ai/v2/openapi/task", {
    method: "POST",
    headers: { "Authorization": `Bearer ${TRIPO_API_KEY}`, "Content-Type": "application/json" },
    body: JSON.stringify({ type: "text_to_model", prompt: prompt })
  });
  const data = await res.json();
  return data.data.task_id; // يرجع لك موديل 3D كامل مفروش
}

// ================== 2. المخ اللي يفهم 3 اوضاع ==================
function buildPrompt(mode, userText) {
  if (mode === 'auto') return `Design a stunning, world-class luxury villa, Saudi modern architecture, fully furnished, pool, garden, 8k, unreal engine, ${userText}`;
  if (mode === 'level') return `Design a ${userText} villa, Saudi style, fully furnished interior, pool, garden, 8k`;
  if (mode === 'manual') return `Generate a 3D villa with this exact spec: ${userText}. Do not add anything else.`;
  return userText;
}

// ================== 3. الواجهة النهائية = مربع واحد + 3 خيارات ==================
export default function App() {
  const [mode, setMode] = useState('auto');
  const [prompt, setPrompt] = useState('');
  const [taskId, setTaskId] = useState(null);
  const [modelUrl, setModelUrl] = useState(null);
  const [loading, setLoading] = useState(false);

  const handleDesign = async () => {
    setLoading(true);
    const finalPrompt = buildPrompt(mode, prompt);
    const id = await generate3DFromText(finalPrompt);
    setTaskId(id);
    // هنا ننتظر الـ API يخلص وناخذ رابط الموديل .glb
    // setModelUrl("رابط_الموديل.glb") 
    setLoading(false);
  };

  return (
    <div style={{width: '100vw', height: '100vh', background:'#0f172a'}}>
      {!modelUrl ? (
        <div style={{position:'absolute', top:'50%', left:'50%', transform:'translate(-50%,-50%)', width:'500px', color:'white', textAlign:'center'}}>
          <h1 style={{fontSize:'32px'}}>Housin3D AI Architect V4.0 FINAL</h1>
          <p>مافي رجعة. هذا الكامل.</p>
          
          <select value={mode} onChange={(e)=>setMode(e.target.value)} style={{width:'100%', padding:'12px', margin:'10px 0', background:'#1e293b', color:'white', border:'1px solid #334155'}}>
            <option value="auto">1. أوتو: سوي على كيفك شي عالمي</option>
            <option value="level">2. مستوى: اكتب فخم / متوسط / عادي</option>
            <option value="manual">3. يدوي: امر غرفة غرفة بالتفصيل</option>
          </select>

          <textarea 
            value={prompt}
            onChange={(e)=>setPrompt(e.target.value)}
            placeholder={mode==='manual' ? "مثال: المجلس كلاسيك كنبة بني ثريا ذهب. الصالة مودرن كنبة رمادي..." : "مثال: فيلا سعودية فخمة بمسبح وحديقة"}
            style={{width:'100%', height:'120px', padding:'12px', background:'#1e293b', color:'white', border:'1px solid #334155'}}
          />
          <button onClick={handleDesign} disabled={loading} style={{width:'100%', padding:'15px', background:'#06b6d4', color:'black', fontWeight:'bold', border:'none', marginTop:'10px'}}>
            {loading ? 'قاعد يصمم...' : 'صمم لي'}
          </button>
          {taskId && <p>رقم المهمة: {taskId} - انتظر 30 ثانية</p>}
        </div>
      ) : (
        <Canvas camera={{ position: [15, 8, 15] }}>
          <ambientLight intensity={0.6} />
          <directionalLight position={[10, 10, 5]} intensity={1} />
          <Environment preset="sunset" />
          {/* هنا نحمل الموديل اللي رجع من الـ AI */}
          {/* <primitive object={useGLTF(modelUrl).scene} /> */}
          <ContactShadows />
          <OrbitControls enableZoom enablePan />
        </Canvas>
      )}
    </div>
  );
}
