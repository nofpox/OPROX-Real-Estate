import React, { useState } from 'react';

export default function App() {
  const [mode, setMode] = useState('auto');
  const [prompt, setPrompt] = useState('');
  const [result, setResult] = useState('');
  const [loading, setLoading] = useState(false);

  const brain = `
انت مهندس معماري AI اسمه Housin3D. تتكلم بأدب واحتراف مع العميل.

قانون التعامل:
1. اذا العميل قال: "وش عندك؟ عطني افكارك وش رايك ؟"
    = رد عليه: "تفضل، جهزت لك 3 نماذج للاختيار: النموذج الخليجي، النموذج العربي، والنموذج العالمي. اي واحد تفضل؟"
2. اذا العميل قال: "ما عجبني"
    = رد عليه: "ولا يهمك. بعطيك 3 تصاميم جديدة فوراً."
3. اذا العميل اعطاك تفاصيل "كذا كذا كذا"
    = رد عليه: "تم. سأنفذ طلبك بالتفصيل بدون أي تغيير."

ممنوع الكلام العامي. الكلام كله محترف وراقي. انت معلم، مو صديق استراحة.
`;

  const API_KEY = "sk-proj-حط_مفتاح_OpenAI_هنا";

  const buildPrompt = () => {
    if (mode === 'auto') return `${brain}\n\nالعميل قال: وش عندك؟ عطني افكارك وش رايك ؟`;
    if (mode === 'level') return `${brain}\n\nالعميل قال: ${prompt}. قدم 3 نماذج.`;
    return `${brain}\n\nالعميل قال: ${prompt}. نفذ بالحرف.`;
  };

  const generate3DFromText = async () => {
    setLoading(true);
    setResult('');
    try {
      const response = await fetch("https://api.openai.com/v1/chat/completions", {
        method: "POST",
        headers: { "Content-Type": "application/json", "Authorization": `Bearer ${API_KEY}` },
        body: JSON.stringify({
          model: "gpt-4o",
          messages: [{ role: "user", content: buildPrompt() }],
        }),
      });
      const data = await response.json();
      setResult(data.choices[0].message.content);
    } catch (e) {
      setResult("خطأ: " + e.message);
    }
    setLoading(false);
  };

  return (
    <div style={{ padding: 20, fontFamily: 'sans-serif', maxWidth: 700, margin: 'auto' }}>
      <h1>Housin3D AI</h1>
      <div style={{ display: 'flex', gap: 10, marginBottom: 20 }}>
        <button onClick={() => setMode('auto')} style={{ background: mode === 'auto'? 'black' : 'gray', color: 'white', padding: 10 }}>Auto</button>
        <button onClick={() => setMode('level')} style={{ background: mode === 'level'? 'black' : 'gray', color: 'white', padding: 10 }}>Level</button>
        <button onClick={() => setMode('manual')} style={{ background: mode === 'manual'? 'black' : 'gray', color: 'white', padding: 10 }}>Manual</button>
      </div>
      {(mode === 'level' || mode === 'manual') && (
        <textarea value={prompt} onChange={(e) => setPrompt(e.target.value)} placeholder="اكتب طلبك هنا..." style={{ width: '100%', height: 100, marginBottom: 10 }} />
      )}
      <button onClick={generate3DFromText} disabled={loading} style={{ width: '100%', padding: 15, background: 'black', color: 'white' }}>
        {loading? 'جاري التصميم...' : 'صمم الآن'}
      </button>
      {result && <pre style={{ marginTop: 20, whiteSpace: 'pre-wrap', background: '#f0f0f0', padding: 15 }}>{result}</pre>}
    </div>
  );
}
